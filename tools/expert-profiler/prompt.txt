import asyncio
import json
from dataclasses import dataclass
from typing import Optional

@dataclass
class FanState:
    fan_id: int
    rpm: int
    pwm: int
    percent: Optional[float]

class FanController:
    def __init__(self, cli_path: str):
        self.cli_path = cli_path
        self._lock = asyncio.Lock()

    async def read_state(self, fan_id: int) -> FanState:
        async with self._lock:
            proc = await asyncio.create_subprocess_exec(
                self.cli_path, "-f", str(fan_id), "-r",
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
            )
            out, err = await proc.communicate()
            data = json.loads(out.decode())
            return FanState(fan_id, data["rpm"], data["pwm"], data.get("percent"))

    async def set_pwm(self, fan_id: int, value: int) -> None:
        if value < 0 or value > 255:
            raise ValueError("pwm out of range")
        async with self._lock:
            proc = await asyncio.create_subprocess_exec(
                self.cli_path, "-f", str(fan_id), "-v", str(value)
            )
            await proc.wait()
